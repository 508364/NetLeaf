# TLS 扩展模块构建脚本
# 内置 mbedTLS（2.28.x LTS）集成 + TLS/SSL 加密扩展
# 目标产物：libnetleaf_tls.dll/.so

message(STATUS "Building tls module...")

# 使用 third-party 目录中内置的 mbedTLS（2.28.x LTS）。
# 选择 mbedTLS 2.28 是因为它是最后一个仍实现 TLS 1.0/1.1 的 LTS 系列
# （mbedTLS 3.x 已从协议核心中彻底移除 TLS 1.0/1.1）。
set(MBEDTLS_SOURCE_DIR "${CMAKE_SOURCE_DIR}/third-party/mbedtls" CACHE PATH "Path to mbedTLS source directory")
message(STATUS "TLS Extension: Using built-in mbedTLS from ${MBEDTLS_SOURCE_DIR}")

# 将 mbedTLS 作为子目录引入
if(EXISTS "${MBEDTLS_SOURCE_DIR}/CMakeLists.txt")
    # 仅构建 mbedTLS 静态库，并关闭其测试与示例程序
    # （在 add_subdirectory 之前设为缓存项，避免 mbedTLS 自身的 option() 重新开启它们）。
    set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "Build mbedTLS static libraries" FORCE)
    set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "Build mbedTLS shared libraries" FORCE)
    set(ENABLE_TESTING OFF CACHE BOOL "Build Mbed TLS tests" FORCE)
    set(ENABLE_PROGRAMS OFF CACHE BOOL "Build Mbed TLS programs" FORCE)
    set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "Compiler warnings treated as errors" FORCE)

    # mbedTLS 以静态库形式被链入 netleaf_tls 共享库，必须启用位置无关代码（PIC）。
    # 否则 x86_64 链接 .so 时会因 R_X86_64_PC32 重定位失败
    # （"recompile with -fPIC"）；其他架构通常容忍非 PIC，故此前仅 amd64 失败。
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)

    # 开启 mbedTLS 2.28 的实验性 TLS 1.3 支持，使 TLS 扩展覆盖 TLS 1.0 - TLS 1.3。
    # 用户配置头文件在构建树中生成，避免修改内置源码。
    set(_nl_mbedtls_user_cfg "${CMAKE_BINARY_DIR}/netleaf_mbedtls_user_config.h")
    file(WRITE "${_nl_mbedtls_user_cfg}"
         "/* Generated by the NetLeaf build system. */\n"
         "/* Enable TLS 1.3 in mbedTLS 2.28 (experimental, client side). */\n"
         "#define MBEDTLS_SSL_PROTO_TLS1_3_EXPERIMENTAL\n"
         "/* mbedTLS 2.28's timing.c falls back to gettimeofday() on architectures\n"
         " * without an inline-assembly hardclock (e.g. ARM64 on Windows), but it\n"
         " * only includes <sys/time.h> on the Unix path. Make the declaration\n"
         " * visible so the file compiles on Windows as well. */\n"
         "#if defined(_WIN32)\n"
         "#include <sys/time.h>\n"
         "#endif\n")
    set(MBEDTLS_USER_CONFIG_FILE "${_nl_mbedtls_user_cfg}" CACHE FILEPATH "mbedTLS user config" FORCE)

    add_subdirectory("${MBEDTLS_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/mbedtls")
else()
    message(FATAL_ERROR "mbedTLS source not found at ${MBEDTLS_SOURCE_DIR}. Please download mbedTLS v2.28.10 and place it in third-party/mbedtls/")
endif()

# TLS 扩展源文件（纯 TLS 扩展，不依赖 MQTT）
add_library(netleaf_tls SHARED
    netleaf_tls.c
)
set_target_properties(netleaf_tls PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    VERSION ${PROJECT_VERSION}
    SOVERSION ${PROJECT_VERSION_MAJOR}
    OUTPUT_NAME "netleaf_tls"
)

target_include_directories(netleaf_tls PUBLIC
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/src/mqtt
    ${MBEDTLS_SOURCE_DIR}/include
)

target_compile_definitions(netleaf_tls PRIVATE
    NL_TLS_EXPORTS
    NL_TLS_ENABLE
)

target_link_libraries(netleaf_tls PRIVATE
    netleaf_core
    netleaf_lang
    mbedtls
    mbedx509
    mbedcrypto
)

# 平台相关链接
if(WIN32)
    target_link_libraries(netleaf_tls PRIVATE ws2_32)
elseif(UNIX AND NOT APPLE)
    target_link_libraries(netleaf_tls PRIVATE pthread)
endif()

# 安装规则
install(TARGETS netleaf_tls
    ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib
    RUNTIME DESTINATION bin
)
install(FILES
    ${CMAKE_SOURCE_DIR}/include/netleaf_tls.h
    ${CMAKE_SOURCE_DIR}/include/netleaf_mqtt_tls.h
    DESTINATION include/netleaf
)

# 安装 mbedTLS 头文件
install(DIRECTORY "${MBEDTLS_SOURCE_DIR}/include/mbedtls"
        DESTINATION include/mbedtls
        FILES_MATCHING PATTERN "*.h")
